cmake_minimum_required(VERSION 3.24)
project(qmat_bench LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
include(CTest)
add_executable(qmat_host_test host_test.cpp)
add_test(NAME packing_and_input COMMAND qmat_host_test)
add_executable(qmat_apply_host application_host_test.cpp)
option(QMAT_ENABLE_CUDA "Build the CUDA/cuBLAS benchmark" ON)
if(QMAT_ENABLE_CUDA)
  if(NOT DEFINED CMAKE_CUDA_ARCHITECTURES)
    set(CMAKE_CUDA_ARCHITECTURES 80)
  endif()
  enable_language(CUDA)
  find_package(CUDAToolkit 11.0 REQUIRED)
  add_executable(qmat_cuda benchmark.cu)
  set_target_properties(qmat_cuda PROPERTIES CUDA_STANDARD 17 CUDA_STANDARD_REQUIRED ON)
  target_link_libraries(qmat_cuda PRIVATE CUDA::cudart CUDA::cublas)
  add_test(NAME cuda_numerical_validation COMMAND qmat_cuda --self-test)
  add_executable(qmat_apply application.cu)
  set_target_properties(qmat_apply PROPERTIES CUDA_STANDARD 17 CUDA_STANDARD_REQUIRED ON)
  target_link_libraries(qmat_apply PRIVATE CUDA::cudart CUDA::cublas)
endif()
